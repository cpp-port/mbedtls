cmake_minimum_required(VERSION 3.20)
project(ca2_mbedtls LANGUAGES C)

set(CMAKE_POLICY_DEFAULT_CMP0077 NEW)

# All customization lives in this wrapper; upstream remains pinned and unmodified.
set(ENABLE_PROGRAMS OFF)
set(ENABLE_TESTING OFF)
set(USE_STATIC_MBEDTLS_LIBRARY ON)
set(USE_SHARED_MBEDTLS_LIBRARY OFF)
set(USE_STATIC_TF_PSA_CRYPTO_LIBRARY ON)
set(USE_SHARED_TF_PSA_CRYPTO_LIBRARY OFF)
set(MBEDTLS_FATAL_WARNINGS OFF)
set(GEN_FILES ON)
set(DISABLE_PACKAGE_CONFIG_AND_INSTALL ON)
set(TF_PSA_CRYPTO_USER_CONFIG_FILE "${CMAKE_CURRENT_SOURCE_DIR}/configuration/crypto_config.h" CACHE FILEPATH "" FORCE)
if(__REACTOS__)
    add_compile_definitions(__REACTOS__ _WIN32_WINNT=0x0501 WINVER=0x0501)
endif()
add_subdirectory(mbedtls)
add_library(ca2_mbedtls INTERFACE)
target_link_libraries(ca2_mbedtls INTERFACE mbedtls mbedx509 tfpsacrypto)
add_library(ca2::mbedtls ALIAS ca2_mbedtls)
if(__REACTOS__)
    target_sources(tfpsacrypto PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/xp_random.c")
    foreach(target IN ITEMS mbedtls mbedx509 tfpsacrypto)
        target_compile_definitions(${target} PUBLIC __REACTOS__ _WIN32_WINNT=0x0501 WINVER=0x0501)
        # Upstream adds bcrypt unconditionally on Windows. The XP RNG hook replaces it.
        foreach(property IN ITEMS LINK_LIBRARIES INTERFACE_LINK_LIBRARIES)
            get_target_property(libraries ${target} ${property})
            if(libraries)
                list(REMOVE_ITEM libraries bcrypt)
                set_property(TARGET ${target} PROPERTY ${property} "${libraries}")
            endif()
        endforeach()
    endforeach()
    target_link_libraries(tfpsacrypto PUBLIC advapi32)
endif()
